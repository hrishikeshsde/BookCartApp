-- LEGACY: the database schema is now defined by the EF Core migrations in BookCart/Migrations.
--
--   * New database:       create an empty database, then   dotnet ef database update --project BookCart
--                         (set BOOKCART_EF_CONNECTION to the connection string). It also inserts the user types and categories.
--   * Existing database: (created from this script) back it up, run DBScript/B4_adopt_migrations.sql once, then apply the
--                         migrations. See docs/B4_DATABASE_CHANGES.md.
--
-- This script is kept because it is exactly how every existing database was created. The upgrade tests
-- (BookCart.Tests/ExistingDatabaseTests.cs) run it to prove that such a database ends up identical to a new one.
-- Do not edit it: change the model and add a migration instead (dotnet ef migrations add <Name> --project BookCart).
-- The first line used to be a Scaffold-DbContext command; the model is now maintained by hand.

CREATE TABLE UserMaster(  
UserID int IDENTITY(1,1) PRIMARY KEY,  
FirstName varchar(20) NOT NULL ,
LastName varchar(20) NOT NULL,
Username varchar(20) NOT NULL,
Password varchar(40) NULL, -- legacy plaintext column, emptied as users log in (see B0.2_password_hash.sql)
PasswordHash varchar(256) NULL,
Gender varchar(6) NOT NULL,
UserTypeID int NOT NULL,   
)  
GO

CREATE UNIQUE INDEX UX_UserMaster_Username ON UserMaster(Username);
GO

-- No admin account is seeded: passwords are stored hashed, so a plaintext seed row cannot work.
-- Register a user through the app, then promote it:  UPDATE UserMaster SET UserTypeID = 1 WHERE Username = '<username>';

CREATE TABLE Book(
BookID int IDENTITY(1,1) PRIMARY KEY,  
Title varchar(100) NOT NULL ,  
Author varchar(100) NOT NULL ,  
Category varchar(20) NOT NULL ,  
Price decimal(10, 2) NOT NULL ,
CoverFileName varchar(100) NULL
)
GO
---------------------------------------------------------------------------
CREATE TABLE Cart(
CartId varchar(36) PRIMARY KEY ,
UserID int NOT NULL,
DateCreated datetime NOT NULL
)
GO
---------------------------------------------------------------------------
CREATE TABLE CartItems(
CartItemId int IDENTITY(1,1) Primary Key,
CartId varchar(36) NOT NULL,
ProductId int NOT NULL,
Quantity int NOT NULL,
)
GO
---------------------------------------------------------------------------
CREATE TABLE CustomerOrders(
OrderId varchar(20) PRIMARY KEY ,
UserID int NOT NULL,
DateCreated datetime NOT NULL,
CartTotal Decimal(10,2) NOT NULL
)
GO
---------------------------------------------------------------------------
CREATE TABLE CustomerOrderDetails(
OrderDetailsId int IDENTITY(1,1) Primary Key,
OrderId varchar(20) NOT NULL,
ProductId int NOT NULL,
Quantity int NOT NULL,
Price decimal(10,2) NOT NULL
)
GO
---------------------------------------------------------------------------
CREATE TABLE Wishlist(
WishlistId varchar(36) PRIMARY KEY,
UserID int NOT NULL,
DateCreated datetime NOT NULL
)
GO
---------------------------------------------------------------------------
CREATE TABLE WishlistItems(
WishlistItemId int IDENTITY(1,1) Primary Key,
WishlistId varchar(36) NOT NULL,
ProductId int NOT NULL
)
GO
